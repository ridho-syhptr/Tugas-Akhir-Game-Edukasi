import flash.events.MouseEvent;

// Pastikan tombol panah kembali Anda memiliki Instance Name: btn_back_about
button_back_about.addEventListener(MouseEvent.CLICK, kembaliDariAbout);

function kembaliDariAbout(e:MouseEvent):void {
	// Putar SFX Klik (beserta penyesuaian volume dari setting)
	var suaraKlik:SFXKlik = new SFXKlik();
    suaraKlik.play(0, 0, Object(root).sfxTransform);
	
	// Langsung pindah ke Main Menu TANPA mematikan BGM
    gotoAndStop("Main_Menu");
}
