stop();

// 1. SETUP AUDIO GLOBAL (ANTI DOUBLE AUDIO)
// ==========================================

// MATIKAN SEMUA MUSIK YANG MUNGKIN MASIH MENYALA DARI FRAME LAIN
if (Object(root).channelBGM != null) {
    Object(root).channelBGM.stop();
}

// SETUP VARIABEL SETTING VOLUME (JIKA APLIKASI BARU DIBUKA)
if (Object(root).volBGM == null) {
    Object(root).volBGM = 0.5; // Default 50%
    Object(root).volSFX = 0.5; // Default 50%
    
    Object(root).bgmTransform = new SoundTransform(Object(root).volBGM);
    Object(root).sfxTransform = new SoundTransform(Object(root).volSFX);
}

// PUTAR MUSIK MAIN MENU DENGAN KONTROL VOLUME YANG BENAR
if (Object(root).isBGMMenuPlaying == null || Object(root).isBGMMenuPlaying == false) {
    var bgmMenuUtama:MusikBGM = new MusikBGM();
    
    // Pastikan musik masuk ke channelBGM HANYA SATU KALI dengan membawa bgmTransform
    Object(root).channelBGM = bgmMenuUtama.play(0, 999, Object(root).bgmTransform);
    Object(root).isBGMMenuPlaying = true;
}

// --- FUNGSI NAIK TURUN BGM (10% / 0.1) ---
function naikkanBGM(e:MouseEvent):void {
    if (Object(root).volBGM < 1.0) { 
        Object(root).volBGM += 0.1;
        
        // Atasi error presisi desimal Flash (misal 0.3000000004)
        Object(root).volBGM = Math.round(Object(root).volBGM * 10) / 10; 
        terapkanVolumeBGM();
    }
}

function turunkanBGM(e:MouseEvent):void {
    if (Object(root).volBGM > 0.0) { 
        Object(root).volBGM -= 0.1;
        Object(root).volBGM = Math.round(Object(root).volBGM * 10) / 10;
        terapkanVolumeBGM();
    }
}

function terapkanVolumeBGM():void {
    // Buat objek Transform baru agar sistem dipaksa membaca nilai baru
    var transformBaru:SoundTransform = new SoundTransform(Object(root).volBGM);
    Object(root).bgmTransform = transformBaru;
    
    // Terapkan ke lagu utama yang sedang berjalan
    if (Object(root).channelBGM != null) {
        Object(root).channelBGM.soundTransform = Object(root).bgmTransform;
    }
    
    sinkronisasiVisualVolume();
}

// --- FUNGSI NAIK TURUN SFX (10% / 0.1) ---
function naikkanSFX(e:MouseEvent):void {
    if (Object(root).volSFX < 1.0) { 
        Object(root).volSFX += 0.1;
        Object(root).volSFX = Math.round(Object(root).volSFX * 10) / 10;
        terapkanVolumeSFX();
    }
}

function turunkanSFX(e:MouseEvent):void {
    if (Object(root).volSFX > 0.0) { 
        Object(root).volSFX -= 0.1;
        Object(root).volSFX = Math.round(Object(root).volSFX * 10) / 10;
        terapkanVolumeSFX();
    }
}

function terapkanVolumeSFX():void {
    var transformSFXBaru:SoundTransform = new SoundTransform(Object(root).volSFX);
    Object(root).sfxTransform = transformSFXBaru;
    sinkronisasiVisualVolume();
    
    // PENTING: Untuk mengetes volume SFX saat ditekan
    var suaraTes:SFXKlik = new SFXKlik();
    suaraTes.play(0, 0, Object(root).sfxTransform);
}

// --- FUNGSI ANIMASI BALOK VOLUME ---
// Mengatur frame pada movieclip_linebacksound & movieclip_linesoundeffect
function sinkronisasiVisualVolume():void {
    // Konversi desimal (0.1 - 1.0) menjadi nomor frame (asumsi ada 11 frame: 1 kosong, 2-11 isi balok)
    // Silakan sesuaikan jumlah frame di dalam movieclip Anda
    var frameBGM:int = Math.round(Object(root).volBGM * 10) + 1;
    var frameSFX:int = Math.round(Object(root).volSFX * 10) + 1;
    
    popup_setting.movieclip_linebacksound.gotoAndStop(frameBGM);
    popup_setting.movieclip_linesoundeffect.gotoAndStop(frameSFX);
}
