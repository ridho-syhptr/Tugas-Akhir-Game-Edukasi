// Mengatur waktu kuis (misal: 60 detik)
var waktuTersisa:int = 180; 
// Membuat timer yang berdetak setiap 1000 milidetik (1 detik)
var timerKuisSedang:Timer = new Timer(1000, waktuTersisa); 

timerKuisSedang.addEventListener(TimerEvent.TIMER, hitungMundurSedang);
timerKuisSedang.start(); // Menjalankan waktu

function hitungMundurSedang(e:TimerEvent):void {
    waktuTersisa--; // Waktu berkurang 1 setiap detik
    teks_waktu.text = waktuTersisa + " detik";
    
    // Jika waktu habis, paksa pindah ke halaman hasil
    if (waktuTersisa <= 0) {
        timerKuisSedang.stop();
        gotoAndStop("Kuis_Result"); 
    }
}

var skorSedang:int = 0;
var indeksSoalSedang:int = 0;

// Ubah batas soal menjadi 15
var totalSoalDimainkan:int = 15;

// 2. Fungsi Mengacak Urutan Soal (Gamifikasi agar tidak bisa dihafal)
bankSoalSedang.sort(acakSoal);
function acakSoal(a:Object, b:Object):int {
    return Math.floor(Math.random() * 3) - 1;
}

// Fungsi baru untuk mengacak urutan opsi A, B, C
function acakPilihan(a:*, b:*):int {
    return (Math.random() > 0.5) ? 1 : -1;
}

// Sembunyikan feedback saat kuis baru dimulai
mc_feedback.gotoAndStop(1);

// 3. Fungsi Menampilkan Soal ke Layar
function tampilkanSoal():void {
    if (indeksSoalSedang < totalSoalDimainkan && indeksSoalSedang < bankSoalSedang.length) {
        
        // --- MENAMPILKAN NOMOR SOAL ---
        // Akan memunculkan teks contoh: "Soal 1 dari 15"
        teks_nomor_soal.text = "Soal " + (indeksSoalSedang + 1) + " dari " + totalSoalDimainkan;
        
        teks_soal.text = bankSoalSedang[indeksSoalSedang].soal;
        teks_skor.text = "Skor: " + skor;
        
        // Mengacak pilihan ganda
        var daftarPilihan:Array = [bankSoalSedang[indeksSoalSedang].opsiA, bankSoalSedang[indeksSoalSedang].opsiB, bankSoalSedang[indeksSoalSedang].opsiC];
        daftarPilihan.sort(acakPilihan); 
        
        teks_pilihan_a.text = daftarPilihan[0];
        teks_pilihan_b.text = daftarPilihan[1];
        teks_pilihan_c.text = daftarPilihan[2];
        
        // --- RESET TAMPILAN UNTUK SOAL BARU ---
        mc_feedback.gotoAndStop(1); // Hilangkan gambar benar/salah
        
        // Aktifkan kembali tombol agar bisa diklik
        button_a.mouseEnabled = true;
        button_b.mouseEnabled = true;
        button_c.mouseEnabled = true;
        
    } else {
        // Jika soal sudah habis, hentikan dan hapus timer
        timerKuisSedang.stop();
        timerKuisSedang.removeEventListener(TimerEvent.TIMER, hitungMundurSedang); // Tambahkan baris ini
        gotoAndStop("Kuis_Result");
    }
}

// Panggil soal pertama kali
tampilkanSoal();

// 4. Mendaftarkan Event Listener untuk Tombol Jawaban
button_a.addEventListener(MouseEvent.CLICK, cekJawaban);
button_b.addEventListener(MouseEvent.CLICK, cekJawaban);
button_c.addEventListener(MouseEvent.CLICK, cekJawaban);

function cekJawaban(e:MouseEvent):void {
    // --- CEGAH KLIK GANDA ---
    // Matikan tombol sementara agar anak-anak tidak menekan tombol berkali-kali saat jeda
    button_a.mouseEnabled = false;
    button_b.mouseEnabled = false;
    button_c.mouseEnabled = false;
    
	// Memutar SFX Klik saat pemain menekan pilihan jawaban
    var suaraKlik:SFXKlik = new SFXKlik();
    suaraKlik.play();
	
    var teksJawabanPemain:String = "";
    
    if(e.currentTarget == button_a) teksJawabanPemain = teks_pilihan_a.text;
    if(e.currentTarget == button_b) teksJawabanPemain = teks_pilihan_b.text;
    if(e.currentTarget == button_c) teksJawabanPemain = teks_pilihan_c.text;
    
    // Mencocokkan jawaban
    if(teksJawabanPemain == bankSoalSedang[indeksSoalSedang].jawabanBenar) {
        skor += 10; 
        mc_feedback.gotoAndStop(2); // Munculkan gambar Centang Hijau di Frame 2
		
		// --- TAMBAHKAN SUARA BENAR ---
        var suaraBenar:SFXBenar = new SFXBenar();
        suaraBenar.play();
		
    } else {
        mc_feedback.gotoAndStop(3); // Munculkan gambar Silang Merah di Frame 3
		
		// --- TAMBAHKAN SUARA SALAH ---
        var suaraSalah:SFXSalah = new SFXSalah();
        suaraSalah.play();
		
    }
    
    // --- FUNGSI JEDA (DELAY) ---
    // Tunggu selama 1500 milidetik (1,5 detik), lalu jalankan fungsi pindahSoal
    setTimeout(pindahSoal, 1500); 
}

// 5. Fungsi Eksekusi Pindah Soal Setelah Jeda
function pindahSoal():void {
    indeksSoalSedang++; // Lanjut ke antrean soal berikutnya
    tampilkanSoal(); // Segarkan tampilan layar
}

button_back_kuis.addEventListener(MouseEvent.CLICK, keluarKuis);
function keluarKuis(e:MouseEvent):void {
    // 1. Putar SFX Klik
    var suaraKlik:SFXKlik = new SFXKlik();
    suaraKlik.play(0, 0, Object(root).sfxTransform);

    if (timerKuisSedang != null) {
        timerKuisSedang.stop(); 
        timerKuisSedang.removeEventListener(TimerEvent.TIMER, hitungMundurSedang); 
    }
    
    // Langsung pindah ke menu level tanpa mematikan BGM
    gotoAndStop("Kuis_Page_Level"); 
}
