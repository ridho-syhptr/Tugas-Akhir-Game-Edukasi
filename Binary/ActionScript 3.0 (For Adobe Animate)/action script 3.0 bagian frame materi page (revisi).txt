import flash.events.MouseEvent;
import flash.media.Sound;
import flash.media.SoundChannel;

// 1. CEK BGM: Putar Musik Materi
if (Object(root).isBGMMateriPlaying == null || Object(root).isBGMMateriPlaying == false) {
    if (Object(root).channelBGM != null) {
        Object(root).channelBGM.stop(); 
    }
    
    // Memanggil audio langsung tanpa 'var'
    Object(root).channelBGM = (new MusikMateri()).play(0, 999); 
    
    Object(root).isBGMMateriPlaying = true;
    Object(root).isBGMMenuPlaying = false;
    Object(root).isBGMLatihanPlaying = false;
}

// 2. TOMBOL KEMBALI KE MAIN MENU
button_back_materi.addEventListener(MouseEvent.CLICK, kembaliKeMenuDariMateriUtama);
function kembaliKeMenuDariMateriUtama(e:MouseEvent):void {
    (new SFXKlik()).play(0, 0, Object(root).sfxTransform);
    
    // Matikan status agar Main Menu mau memutar lagunya sendiri
    Object(root).isBGMMateriPlaying = false; 
    gotoAndStop("Main_Menu");
}
