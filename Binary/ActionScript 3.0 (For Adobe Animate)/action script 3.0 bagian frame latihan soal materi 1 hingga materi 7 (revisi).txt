// 1. CEK BGM: Putar Musik Latihan
if (Object(root).isBGMLatihanPlaying == null || Object(root).isBGMLatihanPlaying == false) {
    if (Object(root).channelBGM != null) {
        Object(root).channelBGM.stop();
    }
    
    // Memanggil audio langsung tanpa 'var' agar bebas Error 1151
    Object(root).channelBGM = (new MusikLatihan()).play(0, 999);
    
    Object(root).isBGMLatihanPlaying = true;
    Object(root).isBGMMenuPlaying = false; 
    Object(root).isBGMMateriPlaying = false; 
}

// ==========================================
// 1. PENGATURAN VARIABEL & WAKTU (3 MENIT)
// ==========================================
var skorKoinLatihan1:int = 0;
var indeksSoalLatihan1:int = 0;
var waktuLatihan1:int = 180; // 3 menit = 180 detik

var timerLatihan1:Timer = new Timer(1000, waktuLatihan1);
timerLatihan1.addEventListener(TimerEvent.TIMER, hitungMundurLatihan1);
timerLatihan1.start();

function hitungMundurLatihan1(e:TimerEvent):void {
    waktuLatihan1--;
    teks_waktu.text = waktuLatihan1 + " detik";
    
    if (waktuLatihan1 <= 0) {
        selesaiLatihan1();
    }
}

// ==========================================
// 2. BANK SOAL (MATERI 1: BILANGAN CACAH)
// ==========================================
var bankSoalLatihan1:Array = [
    {soal: "Jumlah balok ratusan ada 4, puluhan ada 2, dan satuan ada 5. Bilangan berapakah itu?", jawabanBenar: "425", opsiA: "452", opsiB: "425", opsiC: "245"},

var totalSoalLatihan1:int = bankSoalLatihan1.length;

bankSoalLatihan1.sort(acakSoalLatihan1);
function acakSoalLatihan1(a:Object, b:Object):int {
    return Math.floor(Math.random() * 3) - 1;
}

function acakPilihanLatihan1(a:*, b:*):int {
    return (Math.random() > 0.5) ? 1 : -1;
}

// ==========================================
// 3. FUNGSI MENAMPILKAN SOAL
// ==========================================
mc_feedback.gotoAndStop(1);

function tampilkanSoalLatihan1():void {
    if (indeksSoalLatihan1 < totalSoalLatihan1) {
        
        // PENTING: Penulisan nomor dan isi soal sudah digabung ke satu kotak teks (teks_soal)
        teks_soal.text = "Soal " + (indeksSoalLatihan1 + 1) + " dari " + totalSoalLatihan1 + "\n\n" + bankSoalLatihan1[indeksSoalLatihan1].soal;
        teks_koin.text = skorKoinLatihan1 + " Koin"; 
        
        var daftarPilihan:Array = [bankSoalLatihan1[indeksSoalLatihan1].opsiA, bankSoalLatihan1[indeksSoalLatihan1].opsiB, bankSoalLatihan1[indeksSoalLatihan1].opsiC];
        daftarPilihan.sort(acakPilihanLatihan1);
        
        // Nama instance dikembalikan ke bentuk standar tanpa akhiran _latihan1
        teks_pilihan_a.text = daftarPilihan[0];
        teks_pilihan_b.text = daftarPilihan[1];
        teks_pilihan_c.text = daftarPilihan[2];
        
        mc_feedback.gotoAndStop(1); 
        
        button_a.mouseEnabled = true;
        button_b.mouseEnabled = true;
        button_c.mouseEnabled = true;
    } else {
        selesaiLatihan1();
    }
}

tampilkanSoalLatihan1();

// ==========================================
// 4. LOGIKA JAWABAN & SKOR
// ==========================================
button_a.addEventListener(MouseEvent.CLICK, cekJawabanLatihan1);
button_b.addEventListener(MouseEvent.CLICK, cekJawabanLatihan1);
button_c.addEventListener(MouseEvent.CLICK, cekJawabanLatihan1);

function cekJawabanLatihan1(e:MouseEvent):void {
    button_a.mouseEnabled = false;
    button_b.mouseEnabled = false;
    button_c.mouseEnabled = false;
	
	// Memutar SFX Klik saat pemain menekan pilihan jawaban
    var suaraKlik:SFXKlik = new SFXKlik();
    suaraKlik.play();
	
	var teksJawaban:String = "";
    if(e.currentTarget == button_a) teksJawaban = teks_pilihan_a.text;
    if(e.currentTarget == button_b) teksJawaban = teks_pilihan_b.text;
    if(e.currentTarget == button_c) teksJawaban = teks_pilihan_c.text;
    
    if(teksJawaban == bankSoalLatihan1[indeksSoalLatihan1].jawabanBenar) {
        skorKoinLatihan1 += 10; 
        mc_feedback.gotoAndStop(2); 
		
		// --- TAMBAHKAN SUARA BENAR ---
        var suaraBenar:SFXBenar = new SFXBenar();
        suaraBenar.play();
		suaraBenar.play(0, 0, Object(root).sfxTransform);
		
    } else {
        mc_feedback.gotoAndStop(3); 
		
		// --- TAMBAHKAN SUARA SALAH ---
        var suaraSalah:SFXSalah = new SFXSalah();
        suaraSalah.play();
		suaraSalah.play(0, 0, Object(root).sfxTransform);
		
    }
    
    setTimeout(pindahSoalLatihan1, 1500); 
}

function pindahSoalLatihan1():void {
    indeksSoalLatihan1++;
    tampilkanSoalLatihan1();
}

// ==========================================
// 5. PENYELESAIAN LATIHAN & KIRIM DATA
// ==========================================
function selesaiLatihan1():void {
    if (timerLatihan1 != null) {
        timerLatihan1.stop();
        timerLatihan1.removeEventListener(TimerEvent.TIMER, hitungMundurLatihan1);
    }
    
    Object(root).skorAkhir = skorKoinLatihan1;
    Object(root).levelLatihanAktif = 1;
    
    gotoAndStop("Latihan_Result");
}

// ==========================================
// 6. TOMBOL KELUAR DI TENGAH LATIHAN
// ==========================================
// Pastikan Instance Name tombol panah Anda adalah: btn_back_latihan
btn_back_latihan.addEventListener(MouseEvent.CLICK, keluarLatihan1);

function keluarLatihan1(e:MouseEvent):void {
    // 1. Putar SFX Klik
    var suaraKlik:SFXKlik = new SFXKlik();
    suaraKlik.play(0, 0, Object(root).sfxTransform);

    // 2. Matikan Timer agar tidak bocor
    if (timerLatihan1 != null) {
        timerLatihan1.stop();
        timerLatihan1.removeEventListener(TimerEvent.TIMER, hitungMundurLatihan1);
    }
    
    // 3. PENTING: Matikan BGM Latihan dan Reset Statusnya!
    if (Object(root).channelBGM != null) {
        Object(root).channelBGM.stop();
    }
    Object(root).isBGMLatihanPlaying = false; // Memberi tahu sistem bahwa lagu telah usai
    
    // 4. Kembali ke halaman persiapan latihan
    gotoAndStop("Latihan1_Page"); 
}
